cmake_minimum_required(VERSION 3.18)
project(group_ricochet_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Test Gui Qml Quick Widgets)
enable_testing()
add_library(group_core STATIC
    ../../src/ricochet-refresh-qt/groups/GroupManager.cpp
    ../../src/ricochet-refresh-qt/groups/GroupManager.h)
target_link_libraries(group_core PUBLIC Qt5::Core)
target_include_directories(group_core PUBLIC ../../src/ricochet-refresh-qt/groups)
if(MSVC)
    target_compile_options(group_core PRIVATE /utf-8 /W4)
endif()
add_executable(group_tests group_tests.cpp)
target_link_libraries(group_tests PRIVATE group_core Qt5::Test)
add_test(NAME group_core COMMAND group_tests)
add_executable(group_ui_smoke ui_smoke.cpp)
target_link_libraries(group_ui_smoke PRIVATE group_core Qt5::Widgets Qt5::Qml Qt5::Quick)
target_compile_definitions(group_ui_smoke PRIVATE GROUP_QML_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../../src/ricochet-refresh-qt/ui/GroupsWindow.qml")
add_test(NAME group_qml COMMAND group_ui_smoke)
set_tests_properties(group_qml PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
set_tests_properties(group_core group_qml PROPERTIES TIMEOUT 20)
